 def read_data(spark,config):
 
   try :
   
     df=spark.read
	 .format(config["source"]["format"])
	 .option("header",config["source"]["header"])
	 .option("inferschema",config["source"]["inferschema"])
	 .option("path",config["source"]["file_path"])
	 .load()
	 
	 print("data loaded successfully")
	 
	 return df 
	 
	except Exception e:
	
	  print(" Error in Reading Data")
	  
	  print(e)
	  

	 
Transformer.py :
-----------------

 def drop_duplicates(df,config):
 
    if(config["transformation"]["drop_duplicates"]):
        df1=df.dropduplicates()
		
		return df1
	 


def upper_case_columns(df,config):
  
    cols=config["transformation"]["uppercase_columns"]
	
	for c in cols:
	   df2=df.withColumn(
	      c,upper(col(c))
		)
		
	 return df2
	 

def filter_Salary(df,config):

   slary_value=config["Transformation"]["Filter_condition"]["salary"]
   
   df3=df.filter(col("salary")>slary_value)
   
   return df3.
   
   
 def transform_data(df,config):
 
  try:
  
    drop_duplicates(df,config)
	upper_case_columns(df,config)
	filter_Salary(df,config)

 except Exception e:
	
	  print(" Error in Transformation")
	  
	  print(e)
